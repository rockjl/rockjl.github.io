fn check_filter_fields<'a>(fields: &'a syn::punctuated::Punctuated<syn::Field, syn::Token![,]>, attr_str: &str) -> syn::Result<Vec<&'a syn::Field>> {
    let ret = fields.iter().filter(|item|{
        check_and_take_user_specified_from_ident(item, attr_str).unwrap().is_some()
    }).collect();
    Ok(ret)
}

fn get_attribute_string(field: &syn::Field, attr_str: &str) -> Option<syn::LitStr> {
    for attr in &field.attrs {
        if let Ok(syn::Meta::List(syn::MetaList {
            ref path,
            ref nested,
            ..
        })) = attr.parse_meta() {
            if let Some(p) = path.segments.first() {
                if p.ident == attr_str {
                    if let Some(syn::NestedMeta::Lit(syn::Lit::Str(
                        litStr
                    ))) = nested.first() {
                        /*
                        此处：
                            1. 无法创建新的LitStr对象，因为正则表达式前边的r字符无法被正确的转换
                            2. 无法返回引用，因为生命周期属于函数体
                            所以只能克隆一个对象然后返回
                        */
                        return Some(litStr.clone());
                    }
                }
            }
        }
    }
    None
}

fn get_attribute_int(field: &syn::Field, attr_str: &str) -> Option<syn::LitInt> {
    for attr in &field.attrs {
        if let Ok(syn::Meta::List(syn::MetaList {

        })) = attr.parse_meta() {

                if p.ident == attr_str {
                    if let Some(syn::NestedMeta::Lit(syn::Lit::Int(

                        return Some(litInt.clone());
                    }
                }
            }
        }
    }
    None
}

fn get_attribute_tuple(field: &syn::Field, attr_str: &str) -> Option<(syn::LitInt, syn::LitInt)> {
    for attr in &field.attrs {
        if let Ok(syn::Meta::List(syn::MetaList {
            ref path,
            ref nested,
            ..
        })) = attr.parse_meta() {
            if let Some(p) = path.segments.first() {
                if p.ident == attr_str {
                    let mut left_value: syn::LitInt = syn::LitInt::new("0", attr.span());
                    let mut right_value: syn::LitInt = syn::LitInt::new("0", attr.span());
                    if let Some(syn::NestedMeta::Lit(syn::Lit::Int(
                        litInt
                    ))) = nested.first() {
                        left_value = litInt.clone();    
                    };
                    if let Some(syn::NestedMeta::Lit(syn::Lit::Int(
                        litInt
                    ))) = nested.last() {
                        right_value = litInt.clone();    
                    };
                    let ret = (left_value, right_value);
                    return Some(ret);
                }
            }
        }
    }
    None
}

fn generate_email_check(fields: &syn::punctuated::Punctuated<syn::Field, syn::Token![,]>) -> syn::Result<proc_macro2::TokenStream> {
    let emails = check_filter_fields(fields, "Email")?;
    let vec_ts: Vec<proc_macro2::TokenStream> = emails.iter().map(|f| {
        let item_ident = f.ident.as_ref().unwrap();
        let _type = &f.ty;
        quote::quote!(
            let regex = Regex::new(r"^[A-Za-z0-9\u4e00-\u9fa5]+@[a-zA-Z0-9_-]+(\.[a-zA-Z0-9_-]+)+$").unwrap();
            if !regex.is_match(&self.#item_ident) {
                return Err(format!("The Email do not match: {} : {}",stringify!(#item_ident) , r"^[A-Za-z0-9\u4e00-\u9fa5]+@[a-zA-Z0-9_-]+(\.[a-zA-Z0-9_-]+)+$"));
            }
        )
    }).collect();
    let ret = quote::quote!(
        #(#vec_ts)*
    );
    Ok(ret) 
}

fn generate_pattern_check(fields: &syn::punctuated::Punctuated<syn::Field, syn::Token![,]>) -> syn::Result<proc_macro2::TokenStream> {
    let patterns = check_filter_fields(fields, "Pattern")?;
    let vec_ts: Vec<proc_macro2::TokenStream> = patterns.iter().map(|f| {
        let pattern = get_attribute_string(f, "Pattern").unwrap();
        let item_ident = f.ident.as_ref().unwrap();
        let _type = &f.ty;
        quote::quote!(
            let regex = Regex::new(#pattern).unwrap();
            if !regex.is_match(&self.#item_ident) {
                return Err(format!("The Email do not match: {} : {}",stringify!(#item_ident) , #pattern));
            }
        )
    }).collect();
    let ret = quote::quote!(
        #(#vec_ts)*
    );
    Ok(ret) 
}

fn generate_length_check(fields: &syn::punctuated::Punctuated<syn::Field, syn::Token![,]>) -> syn::Result<proc_macro2::TokenStream> {
    let lengths = check_filter_fields(fields, "Length")?;
    let vec_ts: Vec<proc_macro2::TokenStream> = lengths.iter().map(|f| {
        let (left_value, right_value) = get_attribute_tuple(f, "Length").unwrap();
        let item_ident = f.ident.as_ref().unwrap();
        let _type = &f.ty;
        quote::quote!(
            if self.#item_ident.len() > #right_value && self.#item_ident.len() < #left_value {
                return Err(format!("The value exceeds the limit: {} : {} - {}",stringify!(#item_ident) , #left_value, #right_value));
            }
        )
    }).collect();
    let ret = quote::quote!(
        #(#vec_ts)*
    );
    Ok(ret) 
}

fn do_expand_beancheck(st: &syn::DeriveInput) -> syn::Result<proc_macro2::TokenStream> {
    let struct_name = &st.ident;
    let fields = get_fields_from_derive_input(st)?;
    let email_func = generate_email_check(&fields)?;
    let pattern_func = generate_pattern_check(&fields)?;
    let length_func = generate_length_check(&fields)?;
    let range_func = generate_range_check(&fields)?;
    let max_func = generate_max_check(&fields)?;
    let min_func = generate_min_check(&fields)?;
    
    let str = "hello".to_string();
    let ret = quote::quote!(
        impl #struct_name {
            pub fn validate(&self) -> std::result::Result<&#struct_name, std::string::String> {
                #email_func
                #pattern_func
                #length_func
                #range_func
                #max_func
                #min_func
                Ok(self)
            }
        }
    );
    Ok(ret)
}

#[proc_macro_derive(BeanCheck, attributes(Min, Max, Range, Length, Pattern, Email))]
pub fn beancheck_derive(input: TokenStream) -> TokenStream {
    let st = syn::parse_macro_input!(input as syn::DeriveInput);
    match do_expand_beancheck(&st) {
        Ok(token_stream) => token_stream.into(),
        Err(e) => e.to_compile_error().into(),
    }
