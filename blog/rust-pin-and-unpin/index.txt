struct SelfRef {
    value: String,
    ptr: *mut String,
}

impl SelfRef {
    fn new(txt: &str) -> Self {
        SelfRef {
            value: String::from(txt),
            ptr: std::ptr::null_mut(),
        }
    }
    fn init(&mut self) {
        let v_ref = &self.value;
        self.ptr = v_ref as *const String as *mut String;
    }
    fn value(&self) -> &str {
        &self.value
    }
    fn ptr(&self) -> &String {
        unsafe { &*self.ptr }
    }
    fn ptr_address(&self) -> *mut String {
        self.ptr
    }
}

struct SelfRef {
    value: String,
    ptr: *mut String,
}

impl SelfRef {
    fn new(txt: &str) -> Self {
        SelfRef {
            value: String::from(txt),
            ptr: std::ptr::null_mut(),
        }
    }
    fn init(&mut self) {
        let v_ref = &self.value;
        self.ptr = v_ref as *const String as *mut String;
    }
    fn value(&self) -> &str {
        &self.value
    }
    fn value_address(&self) -> *mut String {
        &self.value as *const String as *mut String
    }
    fn value_address2(&self) -> *const u8 {
        self.value.as_ptr()
    }
    fn ptr(&self) -> &String {
        unsafe { &*self.ptr }
    }
    fn ptr_self_address(&self) -> &*mut String {
        &self.ptr as &*mut String
    }
    fn ptr_address(&self) -> *mut String {
        self.ptr
    }
}
