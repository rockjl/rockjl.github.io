    index: i32,
}
fn main() {
	let arc = Arc::new(RwLock::new(GuardMemory{ index: 0 }));
    let mut handles = vec![];
    for idx in 0..10 {
        let arc = arc.clone();
        if idx % 2 == 0 {
            //---创建写线程
            let mut name = "write-thread".to_string();
            name.push_str(idx.to_string().as_str());

            let handle = thread::Builder::new()
            .name(name).spawn(move || {
                let s = thread::current().name().unwrap().to_string();
                println!("{}--准备获得锁", s);
                let mut num = arc.write().unwrap();
                println!("{}++已经获得锁", s);
                thread::sleep(Duration::from_millis(1000 * 1));
                num.index += 1;
                println!("{}++准备释放锁", s);
                drop(num);
                thread::sleep(Duration::from_millis(100));
                println!("{}--已经释放锁", s);
            }).unwrap();
            handles.push(handle);
        } else {
            //---创建读线程
            let mut name = "read-thread".to_string();
            name.push_str(idx.to_string().as_str());

            let handle = thread::Builder::new()
            .name(name).spawn(move || {
                let s = thread::current().name().unwrap().to_string();
                println!("{}--准备获得锁", s);
                let mut num = arc.read().unwrap();
                println!("{}++已经获得锁", s);
                thread::sleep(Duration::from_millis(1000 * 1));
                println!("{}读取数据:{}", s, num.index);
                println!("{}++准备释放锁", s);
                drop(num);
                thread::sleep(Duration::from_millis(100));
                println!("{}--已经释放锁", s);
            }).unwrap();
            handles.push(handle);
        }
    }
    for handle in handles {
        handle.join().unwrap();
    }
    println!("Result: {:?}", arc.read().unwrap());
}
